## Jenkins 설치 명령어 (Ubuntu)

$ docker run -d \
    --name jenkins \
    --restart=always \
    --user root \
    -p 8080:8080 \
    -p 50000:50000 \
    -v jenkins_home:/var/jenkins_home \
    -v /var/run/docker.sock:/var/run/docker.sock \
    jenkins/jenkins:lts-jdk17

## root 확인
$ docker exec jenkins whoami

## 젠킨스 컨테이너 실행
$ docker exec -it -u root jenkins bash

## 실행 확인
$ docker ps

→ Jenkins 컨테이너와 ai-model-api 컨테이너가 보이면 정상

## Jenkins 브라우저 확인
> http://localhost:8080

## 초기 비밀번호 확인
$ docker exec jenkins cat /var/jenkins_home/secrets/initialAdminPassword

> install suggested plugins 선택 → 관리자 계정 생성

## 도커 종료? 삭제?
$ docker rm -f 도커이름

## 젠킨스 컨테이너 안에서
# apt update
# apt install docker.io git curl -y    # 도커, git 플러그인 설치
# docker ps
# exit

## 젠킨스 재시작 명령어
$ docker start jenkins
